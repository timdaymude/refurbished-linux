# Reawakening OS boot menu.
#
# Read by GRUB's gfxmenu as a conventional theme: a dark desktop image, the
# menu list, and a five-second progress bar for the countdown. Nothing here is
# required for booting: if this file or the images are missing, GRUB prints
# "failed to load theme" and falls back to its plain text menu, which is the
# behaviour the images rely on as a safety net.
#
# Fonts are the ones GRUB itself ships (unicode.pf2), so the theme needs no
# font files of its own. The name must match the font's full name exactly:
# GRUB looks fonts up by name, and the unicode.pf2 it ships registers as
# "GNU Unifont Regular 16". An earlier r17 build asked for "Unifont Regular
# 16"; GRUB found no such font, drew the background and progress bar, and
# printed no menu text at all. Keep this in step with the unicode.pf2 in
# grub-common (scripts/check-config.sh verifies it).

desktop-image: "background.png"
desktop-color: "#14161b"
title-text: ""
terminal-font: "GNU Unifont Regular 16"

+ boot_menu {
    left = 20%
    top = 40%
    width = 60%
    height = 40%
    item_font = "GNU Unifont Regular 16"
    item_color = "#a7aebb"
    selected_item_color = "#eceef2"
    item_height = 34
    item_padding = 10
    item_spacing = 4
    icon_width = 0
    icon_height = 0
    selected_item_pixmap_style = "select_*.png"
    scrollbar = false
}

+ progress_bar {
    id = "__timeout__"
    left = 20%
    top = 84%
    width = 60%
    height = 6
    bg_color = "#262b34"
    fg_color = "#35c2a3"
    border_color = "#17a08a"
    text = "Booting in %d seconds"
    text_color = "#737b8a"
    font = "GNU Unifont Regular 16"
}
